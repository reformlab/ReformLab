/** RunIdDisplay component - centralized version (Story 27.12, AC-4). */

import { Copy, Check } from "lucide-react";
import { useState, useCallback } from "react";
import { toast } from "sonner";

const RUN_ID_DISPLAY_LENGTH = 12;

interface RunIdDisplayProps {
  runId: string;
}

export function RunIdDisplay({ runId }: RunIdDisplayProps) {
  const [copied, setCopied] = useState(false);

  // Guard against non-string runId
  const displayId = typeof runId === "string" 
    ? runId.slice(0, RUN_ID_DISPLAY_LENGTH) 
    : String(runId ?? "").slice(0, RUN_ID_DISPLAY_LENGTH);

  const handleCopy = useCallback(async () => {
    try {
      await navigator.clipboard.writeText(runId);
      setCopied(true);
      toast.success("Run ID copied");
      setTimeout(() => setCopied(false), 2000);
    } catch {
      toast.error("Failed to copy");
    }
  }, [runId]);

  return (
    <div className="flex items-center gap-1 shrink-0">
      <span
        className="font-mono text-xs text-slate-500 tabular-nums"
        title={runId}
      >
        {displayId}
      </span>
      <span
        role="button"
        tabIndex={0}
        onClick={handleCopy}
        aria-label={`Copy full run ID: ${runId}`}
        className="text-slate-400 hover:text-slate-600 focus:outline-none focus:ring-2 focus:ring-blue-500 focus:ring-offset-1 rounded p-0.5 cursor-pointer"
        onKeyDown={(e) => {
          if (e.key === "Enter" || e.key === " ") {
            e.preventDefault();
            handleCopy();
          }
        }}
      >
        {copied ? (
          <Check className="h-3 w-3 text-emerald-500" />
        ) : (
          <Copy className="h-3 w-3" />
        )}
      </span>
    </div>
  );
}
