### 2. Fix formatPercent API Ambiguity

**File:** `frontend/src/utils/formatters.ts`
**Issue:** formatPercent(0.44) and formatPercent(44, {multiplyBy100:false}) both return "44%" - ambiguous API

**Corrected code:**
```typescript
/**
 * Format a percentage value for display.
 * ALWAYS multiplies by 100 - input 0.44 becomes "44%".
 * For pre-multiplied values, format as number then append "%" manually.
 * @example formatPercent(0.44) // "44%"
 * @example formatPercent(0.444, { decimals: 1 }) // "44.4%"
 */
export function formatPercent(
  value: number | null | undefined,
  options: Omit<PercentFormatOptions, 'multiplyBy100'> = {}, // Remove multiplyBy100 option
): string {
  if (value === null || value === undefined) return "—";
  if (Number.isNaN(value)) return "NaN";
  if (!Number.isFinite(value)) return value > 0 ? "∞%" : "-∞%";

  const { decimals = 0 } = options;
  const multiplied = value * 100; // Always multiply

  return `${multiplied.toFixed(decimals)}%`;
}
```
